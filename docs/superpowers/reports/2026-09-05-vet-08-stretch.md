Key SQL, verbatim:

```sql
-- decision-path refusals (the silent recorder)
SELECT COUNT(*) FROM decision_records
 WHERE date(timestamp,'-5 hours') BETWEEN '2026-09-02' AND '2026-09-04'
   AND execution_log LIKE '%entry_gate%';                      -- 19 (5 + 14 + 0)
-- era rows and the NULL count (corrected-column law)
SELECT source, COUNT(*), SUM(pnl_corrected IS NULL) FROM trader_positions
 WHERE entry_time >= strftime('%s','2026-08-15 05:00:00')*1000 GROUP BY 1;
-- armed_orders, both clocks normalised
SELECT id, state, state_reason,
       datetime(strftime('%s',created_at),'unixepoch','-5 hours') created_ct,
       datetime(strftime('%s',updated_at),'unixepoch','-5 hours') updated_ct
  FROM armed_orders WHERE id >= 29 ORDER BY id;
-- the broker book at its peak
SELECT id, datetime(emitted_at_ms/1000,'unixepoch','-5 hours'), working_count
  FROM nt8_order_snapshots WHERE working_count >= 5 ORDER BY emitted_at_ms;   -- 1576.. (5), 1583.. (6), 1611 (9)
-- RTH-L all-time
SELECT SUM(outcome='break'), SUM(outcome='hold'), SUM(outcome='ambiguous_horizon')
  FROM touch_outcomes WHERE level_kind='RTH-L';                -- 82 / 42 / 16
-- arm-enablement by side, 09-03 (the moved denominator)
SELECT json_extract(s.value,'$.direction'), SUM(json_extract(s.value,'$.arm.enabled')=1), COUNT(*)
  FROM plans p, json_each(json_extract(p.doc,'$.scenarios')) s
 WHERE p.trade_date='2026-09-03' GROUP BY 1;                   -- long 2/28 · short 8/20
```
